cmake_minimum_required(VERSION 3.20)
project(rntuplizer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# The pinned ROOT version (config/rntuple_env.conf) is passed in by scripts/rnt_build.sh and compiled into the binaries.
set(RNT_PINNED_ROOT "" CACHE STRING "exact ROOT version the binaries must be built and run with")
find_package(ROOT 6.40 CONFIG REQUIRED COMPONENTS Core RIO Tree TreePlayer ROOTNTuple ROOTNTupleUtil)
if(RNT_PINNED_ROOT AND NOT ROOT_VERSION VERSION_EQUAL RNT_PINNED_ROOT)
  message(FATAL_ERROR "found ROOT ${ROOT_VERSION}, pinned version is ${RNT_PINNED_ROOT}")
endif()
foreach(tool rnt_convert rnt_inventory rnt_validate rnt_readbench)
  add_executable(${tool} src/${tool}.cxx)
  target_link_libraries(${tool} PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer ROOT::ROOTNTuple ROOT::ROOTNTupleUtil)
  target_compile_definitions(${tool} PRIVATE RNT_PINNED_ROOT="${RNT_PINNED_ROOT}")
endforeach()
install(TARGETS rnt_convert rnt_inventory rnt_validate rnt_readbench RUNTIME DESTINATION bin)
